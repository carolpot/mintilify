---

The Search Layer exposes two categories of data. **Data Points** are static, verified attributes - contact identity, employment history, emails, phones, and company firmographics. These are refreshed continuously and returned as structured JSON fields on every enrich or search response.

**Signals** are event-driven triggers - headcount changes, job moves, promotions, hiring surges, website traffic shifts, and news events. Signals fire when something meaningful changes and can be polled via the Signals API or delivered via webhooks.

Both types are available across the same four endpoints: `/search`, `/enrich`, `/search-and-enrich`, and `/prospecting` - for both contacts and companies.

export const Chip = ({ label, color }) => {
  const c = color || '#6B7280'
  return (
    <span
      style={{
        display: 'inline-block',
        padding: '0.15rem 0.6rem',
        borderRadius: '999px',
        fontSize: '0.78rem',
        fontWeight: 500,
        color: c,
        background: `${c}1a`,
        border: `1px solid ${c}40`,
        whiteSpace: 'nowrap',
      }}
    >
      {label}
    </span>
  )
}

export const SearchLayerExplorer = () => {
  const [query, setQuery] = useState('')
  const [category, setCategory] = useState('All')
  const [entity, setEntity] = useState('All')
  const [type, setType] = useState('All')
  const [platform, setPlatform] = useState('All')

  const uniq = (values) => Array.from(new Set(values))
  const categories = ['All', ...uniq(searchLayerFields.map((f) => f.category))]
  const entities = ['All', ...uniq(searchLayerFields.map((f) => f.entity))]
  const types = ['All', ...uniq(searchLayerFields.map((f) => f.type))]
  const platforms = ['All', ...uniq(searchLayerFields.flatMap((f) => f.platform))]

  const filtered = searchLayerFields.filter((f) => {
    if (category !== 'All' && f.category !== category) return false
    if (entity !== 'All' && f.entity !== entity) return false
    if (type !== 'All' && f.type !== type) return false
    if (platform !== 'All' && !f.platform.includes(platform)) return false
    if (query) {
      const q = query.toLowerCase()
      if (
        !f.field.toLowerCase().includes(q) &&
        !f.path.toLowerCase().includes(q) &&
        !f.description.toLowerCase().includes(q)
      ) return false
    }
    return true
  })

  const selectStyle = {
    padding: '0.4rem 0.6rem',
    borderRadius: '6px',
    border: '1px solid var(--gray-a6, #d1d5db)',
    fontSize: '0.85rem',
    background: 'var(--background, transparent)',
  }

  return (
    <div>
      <div style={{ display: 'flex', flexWrap: 'wrap', gap: '0.5rem', marginBottom: '0.75rem' }}>
        <input
          type="text"
          placeholder="Search field name, path, or description..."
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          style={{ ...selectStyle, flex: '1 1 220px' }}
        />
        <select value={category} onChange={(e) => setCategory(e.target.value)} style={selectStyle}>
          {categories.map((c) => (
            <option key={c} value={c}>{c === 'All' ? 'All categories' : c}</option>
          ))}
        </select>
        <select value={entity} onChange={(e) => setEntity(e.target.value)} style={selectStyle}>
          {entities.map((c) => (
            <option key={c} value={c}>{c === 'All' ? 'All entities' : c}</option>
          ))}
        </select>
        <select value={type} onChange={(e) => setType(e.target.value)} style={selectStyle}>
          {types.map((c) => (
            <option key={c} value={c}>{c === 'All' ? 'All types' : c}</option>
          ))}
        </select>
        <select value={platform} onChange={(e) => setPlatform(e.target.value)} style={selectStyle}>
          {platforms.map((c) => (
            <option key={c} value={c}>{c === 'All' ? 'All platforms' : c}</option>
          ))}
        </select>
      </div>
      <p style={{ fontSize: '0.85rem', opacity: 0.7, margin: '0 0 0.5rem' }}>
        Showing {filtered.length} of {searchLayerFields.length} fields
      </p>
      <div style={{ overflowX: 'auto' }}>
        <table>
          <thead>
            <tr>
              <th>Field</th>
              <th>Description</th>
              <th>Entity</th>
              <th>Type</th>
              <th>Category</th>
              <th>Path</th>
              <th>Data type</th>
              <th>Refresh</th>
              <th>Platform</th>
            </tr>
          </thead>
          <tbody>
            {filtered.map((f) => (
              <tr key={`${f.category}-${f.field}`}>
                <td><strong>{f.field}</strong></td>
                <td style={{ minWidth: '220px' }}>{f.description}</td>
                <td>{f.entity}</td>
                <td><Chip label={f.type} color={typeColors[f.type]} /></td>
                <td><Chip label={f.category} color={categoryColors[f.category]} /></td>
                <td><code>{f.path}</code></td>
                <td><Chip label={f.dataType} color={dataTypeColors[f.dataType]} /></td>
                <td><Chip label={f.refresh} color={refreshColors[f.refresh]} /></td>
                <td>
                  <div style={{ display: 'flex', flexWrap: 'wrap', gap: '0.3rem' }}>
                    {f.platform.map((p) => (
                      <Chip key={p} label={p} color={platformColors[p]} />
                    ))}
                  </div>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  )
}

<SearchLayerExplorer />
