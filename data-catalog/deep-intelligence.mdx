export const DeepIntelligenceExplorer = () => {
  const [query, setQuery] = useState('')
  const [group, setGroup] = useState('All')
  const [platform, setPlatform] = useState('All')

  const uniq = (values) => Array.from(new Set(values))
  const groups = ['All', ...uniq(deepIntelligenceFeatures.map((f) => f.group))]
  const platforms = ['All', ...uniq(deepIntelligenceFeatures.flatMap((f) => [...f.live, ...f.planned]))]

  const filtered = deepIntelligenceFeatures.filter((f) => {
    if (group !== 'All' && f.group !== group) return false
    if (platform !== 'All' && !f.live.includes(platform) && !f.planned.includes(platform)) return false
    if (query && !f.feature.toLowerCase().includes(query.toLowerCase())) return false
    return true
  })

  const selectStyle = {
    padding: '0.4rem 0.6rem',
    borderRadius: '6px',
    border: '1px solid var(--gray-a6, #d1d5db)',
    fontSize: '0.85rem',
    background: 'var(--background, transparent)',
  }

  const coverageCell = (feature, value) => {
    if (!feature.coverageConfirmed) {
      return <span style={{ opacity: 0.6 }}>Not yet published</span>
    }
    return value
  }

  return (
    <div>
      <div style={{ display: 'flex', flexWrap: 'wrap', gap: '0.5rem', marginBottom: '0.75rem' }}>
        <input
          type="text"
          placeholder="Search feature..."
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          style={{ ...selectStyle, flex: '1 1 220px' }}
        />
        <select value={group} onChange={(e) => setGroup(e.target.value)} style={selectStyle}>
          {groups.map((g) => (
            <option key={g} value={g}>{g === 'All' ? 'All groups' : g}</option>
          ))}
        </select>
        <select value={platform} onChange={(e) => setPlatform(e.target.value)} style={selectStyle}>
          {platforms.map((p) => (
            <option key={p} value={p}>{p === 'All' ? 'All platforms' : p}</option>
          ))}
        </select>
      </div>
      <p style={{ fontSize: '0.85rem', opacity: 0.7, margin: '0 0 0.5rem' }}>
        Showing {filtered.length} of {deepIntelligenceFeatures.length} features
      </p>
      <div style={{ overflowX: 'auto' }}>
        <table>
          <thead>
            <tr>
              <th>Feature</th>
              <th>Group</th>
              <th>Description</th>
              <th>Live on</th>
              <th>Planned</th>
              <th>Endpoint</th>
              <th>Coverage (Global)</th>
              <th>NA</th>
              <th>EMEA</th>
              <th>APAC</th>
            </tr>
          </thead>
          <tbody>
            {filtered.map((f) => (
              <tr key={f.feature}>
                <td><strong>{f.feature}</strong></td>
                <td>{f.group}</td>
                <td>{f.description}</td>
                <td>{f.live.join(', ')}</td>
                <td>{f.planned.length ? f.planned.join(', ') : '-'}</td>
                <td><code>{f.endpoint}</code></td>
                <td>{coverageCell(f, f.coverage.global)}</td>
                <td>{coverageCell(f, f.coverage.na)}</td>
                <td>{coverageCell(f, f.coverage.emea)}</td>
                <td>{coverageCell(f, f.coverage.apac)}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  )
}

<DeepIntelligenceExplorer />

<Info>
  All endpoints are under `https://api.lusha.com` and require an `api_key` header.
</Info>
